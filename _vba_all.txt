Sub 기본_현재슬라이드_모둠지정순서변경_유저폼_변경하기버튼()
        
    ' 변경할 팀 이름을 설정합니다
    OderGroup = OderGroupForm.InputOder.Value
    
    '------------------------------------------------------------------------------------------------
    ' 글자 수 확인 (공백 포함)
    If Len(OderGroup) > 8 Then
        MsgBox "설정할 수 있는 그룹수를 초과하였습니다. 최대 8팀까지 설정할 수 있습니다."
        Exit Sub
    End If
    '------------------------------------------------------------------------------------------------
    
    '------------------------------------------------------------------------------------------------
    ' 1~8까지 한번씩 들어갔는지 확인
    '■■■■■■■■
    Dim a As String
    Dim i As Integer
    Dim counts(1 To 8) As Integer
    Dim isValid As Boolean
    Dim invalidChar As Boolean
    
    ' 변수 a에 테스트할 문자열을 할당
    a = OderGroup
    
    ' 초기값 설정
    isValid = True
    invalidChar = False
    
    ' 각 숫자의 카운트 초기화
    For i = 1 To 8
        counts(i) = 0
    Next i
    
    ' 문자열의 각 문자를 확인
    For i = 1 To Len(OderGroup)
        Dim char As String
        char = Mid(OderGroup, i, 1)
        
        If IsNumeric(char) Then
            Dim num As Integer
            num = CInt(char)
            If num >= 1 And num <= 8 Then
                counts(num) = counts(num) + 1
                If counts(num) > 1 Then
                    isValid = False
                End If
            Else
                invalidChar = True
            End If
        Else
            invalidChar = True
        End If
    Next i
    
    ' 결과 출력
    If isValid And Not invalidChar Then
        'MsgBox "변수 OderGroup는 1부터 4까지의 숫자를 중복 없이 포함하고 있습니다."
    Else
        MsgBox "입력한 데이터의 숫자가 중복되었거나, 유효하지 않은 문자가 포함되어 있습니다. 다시 입력해주세요."
        Exit Sub
    End If
    '------------------------------------------------------------------------------------------------

    
    
    '모둠순서 텍스트 변경
    Slide9.Shapes("(설정)플레이순서지정").TextFrame.TextRange.Text = OderGroup
    
    '이름 변경후 유저폼 자동 닫기
    
    Call 기본_현재슬라이드_모둠지정순서변경_유저폼_닫기
    
End Sub

'소스 파일에서 슬라이드·개체를 가져오는 공용 루틴
Public Sub 기본_데이터가져오기_실행버튼( _
        ByVal sourceFileName As String, _
        Optional ByVal TitlePage As Long = 4, _
        Optional ByVal QuizPage1Start As Long = 71) '■타이틀페이지변수:4 '■1번퀴즈시작페이지변수: 71

    Dim currentPres As Presentation
    Dim sourcePres  As Presentation
    Dim i           As Long
    
    '--- 0. 유효성 검사 ---------------------------------------
    If Trim(sourceFileName) = "" Then
        MsgBox "파일명을 입력해주세요.", vbExclamation
        Exit Sub
    End If
    
    ' 확장자 없으면 .pptm 추가
    If InStrRev(sourceFileName, ".") = 0 Then
        sourceFileName = sourceFileName & ".pptm"
    End If
    
    ' 현재 프레젠테이션과 같은 폴더라면 경로 보정
    If Left(sourceFileName, 1) <> "\" And InStr(sourceFileName, ":") = 0 Then
        sourceFileName = ActivePresentation.Path & "\" & sourceFileName
    End If
    
    '--- 1. 소스 프레젠테이션 열기 ----------------------------
    On Error GoTo 파일열기실패
    Set sourcePres = Presentations.Open(fileName:=sourceFileName, WithWindow:=msoFalse)
    On Error GoTo 0
    
    Set currentPres = ActivePresentation
    
    
    '--- 2. TitlePage(기본 4) 개체 교체 ------------------------
    Dim srcSlide As Slide, tgtSlide As Slide
    Set srcSlide = sourcePres.Slides(TitlePage)
    Set tgtSlide = currentPres.Slides(TitlePage)
    
    For i = tgtSlide.Shapes.count To 1 Step -1
        tgtSlide.Shapes(i).Delete          '기존 개체 모두 삭제
    Next i
    
    srcSlide.Shapes.Range.Copy
    tgtSlide.Shapes.Paste                  '새로 붙여넣기
    
    
   
    
    '--- 3. QuizPage1Start(기본 85) 이후 슬라이드 교체 ----------
Dim srcCount As Long
Dim idx() As Variant
Dim n As Long, j As Long

srcCount = sourcePres.Slides.count

'① 현재 프레젠테이션의 QuizPage1Start 이후 슬라이드 삭제
For i = currentPres.Slides.count To QuizPage1Start Step -1
    currentPres.Slides(i).Delete
Next i

'② 소스 프레젠테이션의 QuizPage1Start~마지막 슬라이드 한꺼번에 복사
If srcCount >= QuizPage1Start Then

    n = srcCount - QuizPage1Start + 1
    ReDim idx(1 To n)

    For j = 1 To n
        idx(j) = QuizPage1Start + j - 1
    Next j

    sourcePres.Slides.Range(idx).Copy
    currentPres.Slides.Paste (currentPres.Slides.count + 1)

End If
    
    
    
    
    
    
    '--- 4. 마무리 --------------------------------------------
    sourcePres.Close
    MsgBox "데이터 가져오기가 완료되었습니다.", vbInformation
    Exit Sub

파일열기실패:
    MsgBox "파일을 열 수 없습니다. 파일명이 정확한지 확인하세요." & vbCrLf & sourceFileName, vbCritical
End Sub

Sub 랜덤슬라이드이동_랜덤숫자생성()
    

    
    C = 15 ' 필요한 문제 초기값 설정

    Dim i As Integer
    Dim j As Integer
    Dim temp As Integer
    Dim isDuplicate As Boolean
    Dim lastSlide As Integer
    Dim QuizCount As Integer
    
    
    lastSlide = ActivePresentation.Slides.count
    QuizCount = ((lastSlide + 1) - 70) / 3 ' 문제은행 총 문제 수 카운트
        
    '--------------------------------------------------------
    '■■■■■■■■■■■ slideIndexes(i) C까지 랜덤
    '기본 문제카드가 C이기에 16까지 랜덤 정의
     
    Randomize
    If QuizCount >= C Then
        ' 문제 수가 15개 이상일 때 중복 제거
        For i = 1 To C
            Do
                temp = Int((C - 1 + 1) * Rnd + 1) ' 1부터 C까지 랜덤 숫자 생성
                isDuplicate = False
                ' 중복된 숫자가 있는지 확인
                For j = 1 To i - 1
                    If slideIndexes(j) = temp Then
                        isDuplicate = True
                        Exit For
                    End If
                Next j
            Loop While isDuplicate ' 중복이 있을 경우 다시 생성
            slideIndexes(i) = temp ' 중복 없는 숫자를 배열에 저장
        Next i
        
        
    Else
        ' 문제 수가 C개 미만일 때는 중복을 허용하여 C개 값 채우기
        For i = 1 To C
            temp = Int((C - 1 + 1) * Rnd + 1) ' 1부터 C까지 랜덤 숫자 생성
            slideIndexes(i) = temp ' 중복된 숫자도 허용하고 배열에 저장
        Next i
    End If

    '--------------------------------------------------------
    '■■■■■■■■■■■ QuizNum(i) 문제 총 수를 확인 후 GotoSlides(i) 이동 슬라이드 인덱스 정의
    Randomize
    lastSlide = ActivePresentation.Slides.count
    QuizCount = ((lastSlide + 1) - 70) / 3 '문제은행 총 문제 수 카운트 ■■■■■문제은행이 ★세팅★설정(70)페이지 부터 시작하며 한 문제당 ★세팅★설정(3)페이지씩 차지
       
    
    If QuizCount >= C Then
         ' 랜덤한 숫자로 배열 짓기 (중복 제거)
        For i = 1 To C 'C개만 랜덤 뽑기
            Do
                temp = Int((QuizCount - 1 + 1) * Rnd + 1)
                isDuplicate = False
                For j = 1 To i - 1
                    If QuizNum(j) = temp Then
                        isDuplicate = True
                        Exit For
                    End If
                Next j
            Loop While isDuplicate
            QuizNum(i) = temp
            GotoSlides(i) = 70 + 1 + ((temp - 1) * 3) '문제슬라이드1~C까지 이동할 슬라이드 번호 정의■■■■■문제은행이 ★세팅★설정(70)페이지, 문제는 ★세팅★설정(70+1) 부터 시작하며 ★세팅★설정(3)페이지씩 차지
        Next i
    
    Else
         ' 랜덤한 숫자로 배열 짓기 (중복 허용)
        For i = 1 To C ' C개만 랜덤 뽑기

Public Sub 랜덤문제카드_설정()
'■이동액션 자체에 랜덤하게 문제선택되도록 하여 해당 기능이 필요없어 미사용.


    Dim slideIndex As Integer
    slideIndex = 10 ' 슬라이드 10의 인덱스 (문제카드가 있는 메인 슬라이드)

    Dim cardIndex As Integer
    Dim hyperlinkIndex As Integer
    Dim hyperlinkSlideNum As Integer
    Dim slideCount As Integer
    slideCount = 15 ' 하이퍼링크로 이동할 슬라이드 개수

    Randomize ' 랜덤 시드 초기화

    Dim Slide As Slide
    Dim cardShape As shape

    ' 슬라이드 10 참조
    Set Slide = ActivePresentation.Slides(slideIndex)


    '■해당기능 미사용■ 슬라이드 10의 "(카드)1"부터 "(카드)15"까지의 개체에 랜덤하게 하이퍼링크 지정
    i = 1
    For cardIndex = 1 To slideCount
        ' "(카드)1"부터 "(카드)15"까지의 개체 참조
        Set cardShape = Slide.Shapes("(카드)" & cardIndex)


        ' 랜덤 슬라이드 번호 하이퍼링크인덱스에 매칭
        hyperlinkIndex = slideIndexes(i)
        i = i + 1

        ' 선택된 슬라이드로 하이퍼링크 지정
        cardShape.ActionSettings(ppMouseClick).Action = ppActionRunMacro
        cardShape.ActionSettings(ppMouseClick).Run = "이동_문제" & hyperlinkIndex

        
        ' 최종 연결 페이지 재구성
        hyperlinkSlideNum = hyperlinkIndex * 3 + 105 '108부터 시작하여 3씩 커짐.
        MsgBox (hyperlinkSlideNum)
        targetSlideName = "Slide" & hyperlinkSlideNum
        ' 선택된 슬라이드로 하이퍼링크 지정
        cardShape.ActionSettings(ppMouseClick).Hyperlink.Address = targetSlideName
    Next cardIndex
    
End Sub
Public Sub 문제은행_문제번호설정()
'■이동액션 자체에 랜덤하게 문제선택되도록 하여 해당 기능이 필요없어 미사용.

Dim slideIndex As Integer
Dim problemNumber As Integer
Dim textBox As shape

' 처음부터(70) 마지막 슬라이드까지 순환
For slideIndex = 70 To ActivePresentation.Slides.count
    ' 현재 슬라이드의 모든 개체를 확인
    For Each textBox In ActivePresentation.Slides(slideIndex).Shapes
        ' 개체의 이름이 "(문제번호)"인지 확인
        If textBox.Name = "(문제번호)" Then
            ' 문제 번호를 증가시키고 텍스트 변경
            problemNumber = problemNumber + 1
            textBox.TextFrame.TextRange.Text = "문제 " & problemNumber
        End If
    Next textBox
Next slideIndex
    

MsgBox ("총 문제 " & problemNumber & "개의 문제번호 타이틀을 자동으로 수정 완료하였습니다.")
    
    
End Sub

Sub 버튼_선택모둠_점수바꾸기()
    
    Dim targetName As String
    Dim sld As Slide
    Dim shp As shape
    Dim shp2 As shape
    Dim grp As shape
    Dim shpName As String
    Dim indices() As Integer
    Dim count As Integer
    Dim tempText As String
    Dim i As Integer

    
    ' 찾을 개체 이름 설정
    targetName = "선택버튼"
    
    ' 초기화
    count = 0
    
    ' 현재 슬라이드 참조
    Set sld = ActivePresentation.SlideShowWindow.View.Slide
    
    ' 슬라이드의 모든 모양 반복
    For Each shp In sld.Shapes
        ' 그룹인 경우
        If shp.Type = msoGroup Then
            ' 그룹 내 개체를 확인하고 개체 이름이 "선택버튼"을 포함하는지 확인
            For Each grp In shp.GroupItems
                If InStr(grp.Name, targetName) > 0 Then
                    ' 개체의 텍스트 값이 "V"인 경우 count 증가
                    
                    If grp.TextFrame.TextRange.Text = "V" Then
                       count = count + 1
                    End If
                End If
            Next grp
        Else
            ' 개별 개체의 경우
            If InStr(shp.Name, targetName) > 0 Then
                ' 개체의 텍스트 값이 "V"인 경우 count 증가
                If shp.TextFrame.TextRange.Text = "V" Then
                   count = count + 1
                End If
            End If
        End If
    Next shp
    
    
    
    
    
    ' V 표시가 2개가 아닌 경우 메시지 표시 후 종료
    If count <> 2 Then
        MsgBox "2개 모둠을 선택해주세요!"
        Exit Sub
    End If
    

    
    ' 인덱스 저장
    ReDim indices(1 To 2)
    i = 1
    ' 슬라이드의 모든 모양 반복
    For Each shp In sld.Shapes
        If shp.Type = msoGroup Then
            ' 그룹 내 개체 확인
            For Each grp In shp.GroupItems
                ' 지정한 문자열이 포함된 개체인지 확인
                If InStr(grp.Name, targetName) > 0 Then
                    ' 개체의 텍스트 값이 "V"인 경우
                    If grp.TextFrame.TextRange.Text = "V" Then
                        ' 인덱스 저장
                        indices(i) = Val(Mid(grp.Name, 2, 1))
                        i = i + 1
                    End If
                End If
            Next grp
        Else
            ' 개별 개체의 경우
            If InStr(shp.Name, targetName) > 0 Then
                ' 개체의 텍스트 값이 "V"인 경우
                If shp.TextFrame.TextRange.Text = "V" Then
                    ' 인덱스 저장
                    indices(i) = Val(Mid(shp.Name, 2, 1))
                    i = i + 1
                End If
            End If
        End If
    Next shp
    
    
    
    ' V가 표시된 개체가 없는 경우 종료
    If i = 1 Then
        MsgBox "V가 표시된 개체가 없습니다."
        Exit Sub
    End If
    
    

    
       
    ' 개체 텍스트 값 교환
            
        
        Set shp = Shapes("(" & indices(1) & "모둠)점수")
        tempText1 = shp.TextFrame.TextRange.Text
         
        Set shp2 = Shapes("(" & indices(2) & "모둠)점수")
        tempText2 = shp2.TextFrame.TextRange.Text
        
        
        shp.TextFrame.TextRange.Text = tempText2
        shp2.TextFrame.TextRange.Text = tempText1

    
    
End Sub

Sub 버튼_게임시작하기()

'----------------------------------------------------------------------------------------------------------------
'■■■ 보안 확인
Call ChkLock_설정확인

If ChkLock = True Then
    Call 보안_문서수정확인
ElseIf ChkLock = False Then
    'MsgBox ("진행함")
End If
   
'------------------------------------
'보안마크 검사 후 매크로 진행여부
Call 보안마크_검사

If ChkWaterMark <> 1 Then
Exit Sub
End If
'------------------------------------

'----------------------------------------------------------------------------------------------------------------
'■■■ 플레이순서 기본세팅


If Slide9.Shapes("(설정)버튼-플레이순서-지정").Fill.ForeColor.RGB = RGB(255, 192, 0) And Slide9.Shapes("(설정)버튼-플레이순서-랜덤").Fill.ForeColor.RGB = RGB(217, 217, 217) Then
   Call 이번차례_초기배열_고정
ElseIf Slide9.Shapes("(설정)버튼-플레이순서-랜덤").Fill.ForeColor.RGB = RGB(255, 192, 0) And Slide9.Shapes("(설정)버튼-플레이순서-지정").Fill.ForeColor.RGB = RGB(217, 217, 217) Then
   Call 이번차례_초기배열_랜덤
Else '둘다 비활성화 또는 선택된경우
   MsgBox ("지정 순서 또는 랜덤 순서 중 하나를 선택해주세요")
   Exit Sub
End If
   
'----------------------------------------------------------------------------------------------------------------
'■■■ 모둠판 설정 모둠수에 맞게 보이기

   ' 타켓 개체의 이름 (shapeName) 설정
    shapeName = "(설정)모둠수"
    GNumbers = Slide9.Shapes(shapeName).TextFrame.TextRange.Text '■개체 이름으로 타켓 개체의 값 정의하기


    '--------------------------------------------------------------
    '모든 모둠 보이기 처리
    For NowGNumber = 1 To 8
        Slide10.Shapes("(" & NowGNumber & "모둠)그룹전체").Visible = True
    Next NowGNumber
    '--------------------------------------------------------------


If GNumbers < 8 Then
    
    '--------------------------------------------------------------
    '설정모둠보다 큰 수의 모둠은 숨기기 처리
    For NowGNumber = GNumbers + 1 To 8
        Slide10.Shapes("(" & NowGNumber & "모둠)그룹전체").Visible = False
    Next NowGNumber
    '--------------------------------------------------------------

ElseIf GNumbers >= 8 Then
    '--------------------------------------------------------------
    '숨기기 처리 개체 없음
    '--------------------------------------------------------------

End If
'----------------------------------------------------------------------------------------------------------------


'--------------------------------------------
'■■■ 퀴즈수가 필요로하는 퀴즈 C보다 작은경우 처리
lastSlide = ActivePresentation.Slides.count
QuizCount = ((lastSlide + 1) - 70) / 3 '문제은행 총 문제 수 카운트 ■■■■■문제은행이 ★세팅★설정(70)페이지 부터 시작하며 한 문제당 ★세팅★설정(3)페이지씩 차지

If QuizCount < 1 Then
MsgBox ("슬라이드에 문제가 없습니다. 최소 문제 슬라이드를 만들어 주세요.")
Exit Sub
Else
End If
'--------------------------------------------



'----------------------------------------------------------------------------------------------------------------
'■■■랜덤슬라이드이동 랜덤숫자 생성
Call 랜덤슬라이드이동_랜덤숫자생성

'----------------------------------------------------------------------------------------------------------------
'■■■랜덤슬라이드이동 랜덤아이템숫자 생성
Call 랜덤슬라이드이동_랜덤아이템숫자생성

'----------------------------------------------------------------------------------------------------------------
'■■■랜덤 문제카드 설정
'Call 랜덤문제카드_설정(미사용: 이동액션자체에 랜덤하여 이동하게 끔 설정)

'----------------------------------------------------------------------------------------------------------------
'■■■문제카드 모두 보이게 초기화
Call 문제카드_모두보이기

'----------------------------------------------------------------------------------------------------------------
'■■■페이지 초기화 설정
Call 초기화_게임시작설정

'----------------------------------------------------------------------------------------------------------------
'■■■크레딧그룹 열기 설정
Call 크레딧_열기



'----------------------------------------------------------------------------------------------------------------

        '----------------------------
        '★★★★모둠별 보상 점수 초기화
        For i = 1 To 8
        NowGroupScore(i) = "0"
        Next i
        '----------------------------


'----------------------------------------------------------------------------------------------------------------
'■■■ 슬라이드 이동
SlideShowWindows(1).View.GotoSlide 10


End Sub

Sub 이번차례_초기배열_랜덤()
    Dim i As Integer
    Dim temp As Integer
    Dim randIndex As Integer
    
    
    '------------------------------------------------------------------------------------------------------------
    ' 타켓 개체의 이름 (shapeName) 설정
    shapeName = "(설정)모둠수"

     '<슬라이드 순환> 'Slide9 이름의 슬라이드 찾기
    For Each mySlide In ActivePresentation.Slides
            If mySlide.Name = "Slide9" Then '■슬라이드 이름으로 타켓 슬라이드 찾기
               GNumbers = mySlide.Shapes(shapeName).TextFrame.TextRange.Text '■개체 이름으로 타켓 개체의 값 정의하기
            Exit For
        End If
    Next mySlide
    '------------------------------------------------------------------------------------------------------------
    
    
    
    
    ' 배열 초기화
    For i = 1 To GNumbers
        myNumbers(i) = i
    Next i
    
    ' 배열 무작위 섞기 (셔플)
    For i = 1 To GNumbers
        randIndex = Int((GNumbers - i + 1) * Rnd + i)
        temp = myNumbers(i)
        myNumbers(i) = myNumbers(randIndex)
        myNumbers(randIndex) = temp
    Next i
    
    ' 인덱스 초기화
    currentIndex = 1
    
    
    
    '------------------------------------------------------------------------------------------------------------
    ' 버튼 활성화 색상 표시
              Slide9.Shapes("(설정)버튼-플레이순서-랜덤").Fill.ForeColor.RGB = RGB(255, 192, 0) '■선택색상으로 변환
              Slide9.Shapes("(설정)버튼-플레이순서-지정").Fill.ForeColor.RGB = RGB(217, 217, 217) '■비활성색상으로 변환
    '------------------------------------------------------------------------------------------------------------

    
End Sub

Sub 버튼_자동점수반영_메인이동()

        Slide10.Shapes("(" & NowGNumber & "모둠)획득점수").TextFrame.TextRange.Text = Replace("+ " & CInt(NowGroupScore(NowGNumber)), vbCrLf, "")

Sub 보안_텍스트_일괄입력()


    Dim ChkText1 As String
    Dim Slide As Slide
    Dim shape As shape
    Dim foundTitle As Boolean
    Dim updatedCount As Integer
    Dim NowSlideIndex As Integer
    
    ' 현재 슬라이드 인덱스 가져오기
    Dim activeSlide As Slide
    Set activeSlide = ActivePresentation.SlideShowWindow.View.Slide
    NowSlideIndex = activeSlide.slideIndex
    
    ' 보안 텍스트 가져오기
    ChkText1 = ActivePresentation.Slides(NowSlideIndex).Shapes("(버튼)보안텍스트").TextFrame.TextRange.Text

    updatedCount = 0
    foundTitle = False
    
    ' 일반 슬라이드 순회
    For Each Slide In ActivePresentation.Slides
        For Each shape In Slide.Shapes
            If shape.Name = "(타이틀체크)텍스트" Then
                shape.TextFrame.TextRange.Text = ChkText1
                updatedCount = updatedCount + 1
                foundTitle = True
            End If
        Next shape
    Next Slide

    ' 모든 디자인의 슬라이드 마스터 및 레이아웃 순회
    Dim design As design
    Dim master As master
    Dim layout As CustomLayout

    For Each design In ActivePresentation.Designs
        Set master = design.SlideMaster

        ' 마스터 자체의 도형들
        For Each shape In master.Shapes
            If shape.Name = "(타이틀체크)텍스트" Then
                shape.TextFrame.TextRange.Text = ChkText1
                updatedCount = updatedCount + 1
                foundTitle = True
            End If
        Next shape

        ' 마스터의 각 레이아웃
        For Each layout In master.CustomLayouts
            For Each shape In layout.Shapes
                If shape.Name = "(타이틀체크)텍스트" Then
                    shape.TextFrame.TextRange.Text = ChkText1
                    updatedCount = updatedCount + 1
                    foundTitle = True
                End If
            Next shape
        Next layout
    Next design

    ' 결과 알림
    If Not foundTitle Then
        MsgBox "(타이틀체크)텍스트 개체가 없습니다.", vbExclamation
    Else
        MsgBox "보안 텍스트 일괄입력이 완료되었습니다." & vbCrLf & "총 입력된 개체 수: " & updatedCount, vbInformation
    End If


End Sub

    originalImages.Add "(보안마크)득템공방로고", True

    On Error Resume Next

    On Error GoTo 0

        ChkWaterMark = -1
        Exit Sub
    End If

    ' 1. 텍스트 개체 검사 (마스터 및 레이아웃 포함)
    For Each shapeName In originalTexts.Keys
        On Error Resume Next
        Set shape = masterSlide.Shapes(shapeName) ' 마스터에서 검색
        On Error GoTo 0

        If shape Is Nothing Then

            For Each layout In masterSlide.CustomLayouts
                On Error Resume Next

                On Error GoTo 0
                If Not shape Is Nothing Then Exit For ' 찾으면 종료
            Next layout
        End If

        ' 개체 검증
        If shape Is Nothing Or shape.TextFrame.TextRange.Text <> originalTexts(shapeName) Then
            isValid = False
        End If
    Next shapeName

    ' 2. 이미지 개체 검사 (마스터 및 레이아웃 포함)
    For Each shapeName In originalImages.Keys

            Next layout

    Next shapeName

    If isValid Then

    Else

Sub 보안마크_검사()
    Dim masterSlide As Object
    Dim layout As CustomLayout
    Dim shape As shape
    Dim shapeName As Variant
    Dim originalTexts As Object, originalImages As Object
    Dim isValid As Boolean
    
    '▼―[새로 추가] : 워터마크 그룹 기준값-----------------------------------
    Const WM_GRP_NAME As String = "(보안마크)그룹"
    Const EXPECTED_LEFT   As Single = 747.8213        '■■■■■워터마크 위치정보' ← 원본 좌표/크기를 넣어주세요
    Const EXPECTED_TOP    As Single = 0               '■■■■■워터마크 위치정보
    Const EXPECTED_WIDTH  As Single = 206.0672        '■■■■■워터마크 위치정보
    Const EXPECTED_HEIGHT As Single = 32.12134        '■■■■■워터마크 위치정보
    Const TOL             As Single = 0.1      ' 허용 오차(포인트)
    '▲--------------------------------------------------------------------
    
    ' 검사 상태 초기화
    ChkWaterMark = 0
    isValid = True
    
    ' 검사 대상 사전
    Set originalTexts = CreateObject("Scripting.Dictionary")
    Set originalImages = CreateObject("Scripting.Dictionary")
    
    originalTexts.Add "(보안마크)PRODUCED BY", "PRODUCED BY"
    originalTexts.Add "(보안마크)YANGSONGI TEACHER'S", "YANGSONGI TEACHER’S"
    originalTexts.Add "(보안마크)LUCKYTEM STUDIO", "LUCKYTEM STUDIO"
    
    originalImages.Add "(보안마크)양송이싸인", True
    originalImages.Add "(보안마크)득템공방로고", True
    
    ' 슬라이드 마스터
    On Error Resume Next
    Set masterSlide = ActivePresentation.Designs(1).SlideMaster
    If masterSlide Is Nothing Then Set masterSlide = ActivePresentation.SlideMaster
    On Error GoTo 0
    
    If masterSlide Is Nothing Then
        MsgBox "오류: 슬라이드 마스터를 찾을 수 없습니다!", vbCritical
        ChkWaterMark = -1
        Exit Sub
    End If
    
    '―[1] 텍스트 검사 -----------------------------------------------------
    For Each shapeName In originalTexts.Keys
        Set shape = Nothing
        On Error Resume Next: Set shape = masterSlide.Shapes(shapeName): On Error GoTo 0
        If shape Is Nothing Then
            For Each layout In masterSlide.CustomLayouts
                On Error Resume Next: Set shape = layout.Shapes(shapeName): On Error GoTo 0
                If Not shape Is Nothing Then Exit For
            Next
        End If
        
        If shape Is Nothing Or shape.TextFrame.TextRange.Text <> originalTexts(shapeName) Then
            isValid = False
        End If
    Next
    
    '―[2] 이미지 검사 -----------------------------------------------------
    For Each shapeName In originalImages.Keys
        Set shape = Nothing
        On Error Resume Next: Set shape = masterSlide.Shapes(shapeName): On Error GoTo 0
        If shape Is Nothing Then
            For Each layout In masterSlide.CustomLayouts
                On Error Resume Next: Set shape = layout.Shapes(shapeName): On Error GoTo 0
                If Not shape Is Nothing Then Exit For
            Next
        End If
        
        If shape Is Nothing Then isValid = False
    Next
    
    '―[3] 그룹 위치/크기 + 최상단으로 끌어올리기 --------------------------
    Dim grp As shape: Set grp = Nothing
    On Error Resume Next
    Set grp = masterSlide.Shapes(WM_GRP_NAME)
    On Error GoTo 0
    
    If grp Is Nothing Then
        isValid = False                     ' 그룹 자체가 없으면 실패
    Else
        ' ① 위치·크기 검증
        If Abs(grp.Left - EXPECTED_LEFT) > TOL _
        Or Abs(grp.Top - EXPECTED_TOP) > TOL _
        Or Abs(grp.Width - EXPECTED_WIDTH) > TOL _
        Or Abs(grp.Height - EXPECTED_HEIGHT) > TOL Then
            isValid = False
        End If
        
        ' ② 최상단 배치 (Z?Order가 1이 아니면 BringToFront)
        If grp.ZOrderPosition <> 1 Then
            grp.ZOrder msoBringToFront
            '※ 굳이 검사에 실패 처리할 필요가 없으므로 isValid는 변경하지 않음
        End If
    End If
    '▲--------------------------------------------------------------
    
    '―[4] 결과 처리 -------------------------------------------------------
    If isValid Then
        ChkWaterMark = 1    ' 정상
    Else
        ChkWaterMark = -1   ' 변조 감지
        Call 보안마크_검사결과_진행여부
    End If
End Sub

Private Sub btnExport_Click()
    Dim thumbSlide As Integer, thumbName As String
    Dim thumbWidth As Long, thumbHeight As Long
    Dim refSlidesText As String, refSlides() As String
    Dim refBaseName As String
    Dim refWidth As Long, refHeight As Long
    Dim exportPath As String
    Dim i As Integer, slideIndex As Integer

    On Error GoTo ErrHandler

    ' 썸네일 값
    thumbSlide = Val(Trim(txtThumbnailSlide.Text))
    thumbName = Trim(txtThumbnailName.Text)
    thumbWidth = CLng(Val(txtThumbWidth.Text))
    thumbHeight = CLng(Val(txtThumbHeight.Text))

    ' 참조 이미지 값
    refSlidesText = Trim(txtRefSlides.Text)
    If refSlidesText = "" Then
        MsgBox "참조 슬라이드 번호를 입력하세요.", vbExclamation
        Exit Sub
    End If

    refSlides = Split(refSlidesText, ",")
    refBaseName = Trim(txtRefBaseName.Text)
    refWidth = CLng(Val(txtRefWidth.Text))
    refHeight = CLng(Val(txtRefHeight.Text))

    ' 저장 경로 확인
    exportPath = ActivePresentation.Path & "\"
    If exportPath = "\" Then
        MsgBox "프레젠테이션을 먼저 저장한 후 다시 시도하세요.", vbExclamation
        Exit Sub
    End If

    ' 썸네일 저장
    If thumbSlide >= 1 And thumbSlide <= ActivePresentation.Slides.count Then
        ActivePresentation.Slides(thumbSlide).Export _
            exportPath & thumbName & ".png", "PNG", thumbWidth, thumbHeight
    Else
        MsgBox "썸네일 슬라이드 번호가 잘못되었습니다.", vbCritical
        Exit Sub
    End If

    ' 참조 이미지 저장 (입력된 개수만큼 자동 처리)
    For i = 0 To UBound(refSlides)
        slideIndex = Val(Trim(refSlides(i)))
        If slideIndex >= 1 And slideIndex <= ActivePresentation.Slides.count Then
            ActivePresentation.Slides(slideIndex).Export _
                exportPath & refBaseName & (i + 1) & ".png", "PNG", refWidth, refHeight
        Else
            MsgBox "참조 슬라이드 번호 " & refSlides(i) & "가 잘못되었습니다.", vbExclamation
            Exit Sub
        End If
    Next i

    MsgBox "모든 이미지가 성공적으로 저장되었습니다!", vbInformation
    
    '창 닫기
    frmExportImages.HIde
       
    Exit Sub

ErrHandler:
    MsgBox "오류 발생: " & Err.Description, vbCritical
End Sub
